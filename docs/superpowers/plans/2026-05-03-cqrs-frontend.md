# CQRS Frontend Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a small `vue/` Vite app that calls the planned CQRS HTTP API through real endpoints, with tabs for users, messages, emails, and a shared request log.

**Architecture:** Create a standalone Vue 3 + TypeScript app under `vue/` using Vue Router for tab routes. Keep endpoint churn isolated inside `src/api/` with one shared fetch client, per-resource request helpers, and a lightweight in-memory request log composable used by all views.

**Tech Stack:** `pnpm`, `vite`, `vue`, `vue-router`, TypeScript, native `fetch`

- [ ] **Step 5: Create shared fetch wrapper**

```ts
import type { ApiError, HttpMethod, RequestLogEntry } from '../types/api'
import { useRequestLog } from '../composables/useRequestLog'

interface RequestOptions {
  method: HttpMethod
  url: string
  body?: unknown
}

const makeError = (message: string, status?: number, responseBody?: unknown): ApiError => {
  const error = new Error(message) as ApiError
  error.status = status
  error.responseBody = responseBody
  return error
}

export async function requestJson<T>({ method, url, body }: RequestOptions): Promise<T> {
  const { pushEntry } = useRequestLog()
  const started = performance.now()
  let responseBody: unknown

  try {
    const response = await fetch(url, {
      method,
      headers: {
        'Content-Type': 'application/json',
      },
      body: body === undefined ? undefined : JSON.stringify(body),
    })

    const text = await response.text()
    responseBody = text ? JSON.parse(text) : null

    const entry: RequestLogEntry = {
      id: crypto.randomUUID(),
      startedAt: new Date().toISOString(),
      method,
      url,
      requestBody: body,
      status: response.status,
      durationMs: Math.round(performance.now() - started),
      responseBody,
    }
    pushEntry(entry)

    if (!response.ok) {
      throw makeError(`HTTP ${response.status}`, response.status, responseBody)
    }

    return responseBody as T
  } catch (error) {
    const normalized = error instanceof Error ? error : new Error('Unknown request error')
    pushEntry({
      id: crypto.randomUUID(),
      startedAt: new Date().toISOString(),
      method,
      url,
      requestBody: body,
      durationMs: Math.round(performance.now() - started),
      responseBody,
      errorMessage: normalized.message,
    })
    throw normalized
  }
}
```

import { apiConfig } from './config'
import { requestJson } from './http'

export interface CreateUserInput {
  email: string
}

export function createUser(input: CreateUserInput) {
  return requestJson<Record<string, unknown>>({
    method: 'POST',
    url: apiConfig.users.create(),
    body: input,
  })
}

export function getUserById(id: string) {
  return requestJson<Record<string, unknown>>({
    method: 'GET',
    url: apiConfig.users.getById(id),
  })
}

<script setup lang="ts">
import { reactive, ref } from 'vue'
import { createUser, getUserById } from '../api/users'
import DetailCard from '../components/DetailCard.vue'
import ErrorAlert from '../components/ErrorAlert.vue'
import FormSection from '../components/FormSection.vue'

const createForm = reactive({
  email: '',
})

const getForm = reactive({
  id: '',
})

const result = ref<Record<string, unknown> | null>(null)
const errorMessage = ref<string | null>(null)
const loading = ref(false)

async function handleCreate() {
  loading.value = true
  errorMessage.value = null

  try {
    result.value = await createUser({ email: createForm.email })
  } catch (error) {
    errorMessage.value = error instanceof Error ? error.message : 'Create user failed'
  } finally {
    loading.value = false
  }
}

async function handleGet() {
  loading.value = true
  errorMessage.value = null

  try {
    result.value = await getUserById(getForm.id)
  } catch (error) {
    errorMessage.value = error instanceof Error ? error.message : 'Get user failed'
  } finally {
    loading.value = false
  }
}
</script>
